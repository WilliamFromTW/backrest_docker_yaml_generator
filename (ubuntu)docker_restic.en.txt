2.
Docker Installation
--
# Add Docker's official GPG key:
sudo apt update
sudo apt install ca-certificates curl
sudo install -m 0755 -d /etc/apt/keyrings
sudo curl -fsSL https://download.docker.com/linux/ubuntu/gpg -o /etc/apt/keyrings/docker.asc
sudo chmod a+r /etc/apt/keyrings/docker.asc

# Add the repository to Apt sources:
sudo tee /etc/apt/sources.list.d/docker.sources <<EOF
Types: deb
URIs: https://download.docker.com/linux/ubuntu
Suites: $(. /etc/os-release && echo "${UBUNTU_CODENAME:-$VERSION_CODENAME}")
Components: stable
Architectures: $(dpkg --print-architecture)
Signed-By: /etc/apt/keyrings/docker.asc
EOF

sudo apt update

sudo apt install docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin

# vi /lib/systemd/system/docker.service (change default Docker data-root directory)
ExecStart=/usr/bin/dockerd --data-root=/docker -H fd:// --containerd=/run/containerd/containerd.sock

systemctl daemon-reload
systemctl enable docker
systemctl restart docker

# Backrest Configuration Note (Official Documentation: https://github.com/garethgeorge/backrest )
# IMPORTANT: Because Backrest runs inside a Docker container, any backup source (source / Plan) and target or restore directory (dest / Repo / Restore)
# configured in http://localhost:9898 MUST use the container paths /data/source and /data/dest rather than the host paths /mnt/source and /mnt/dest:
# - Backup Source (Plan -> Paths to backup): use /data/source/<subfolder_name>  (maps to host /mnt/source/<subfolder_name>)
# - Backup Target / Local Repo or Restore Path (Repo / Restore): use /data/dest/<subfolder_name>  (maps to host /mnt/dest/<subfolder_name>)
